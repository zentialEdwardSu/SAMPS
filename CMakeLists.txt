cmake_minimum_required(VERSION 3.24)
project(SAMPS VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
# ONNX Runtime is loaded at run time by full path (native/src/runtime.cpp), so only its
# headers are needed here. Photoshop ships its own onnxruntime.dll; ours is renamed.
set(SAMPS_ORT_ROOT "${CMAKE_SOURCE_DIR}/.deps/onnxruntime-dml-1.24.4" CACHE PATH "Microsoft.ML.OnnxRuntime.DirectML package")
set(SAMPS_DML_ROOT "${CMAKE_SOURCE_DIR}/.deps/directml-1.15.4" CACHE PATH "Microsoft.AI.DirectML package")
foreach(required "${SAMPS_ORT_ROOT}/build/native/include/onnxruntime_cxx_api.h"
                 "${SAMPS_ORT_ROOT}/runtimes/win-x64/native/onnxruntime.dll"
                 "${SAMPS_DML_ROOT}/include/DirectML.h" "${SAMPS_DML_ROOT}/bin/x64-win/DirectML.dll")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing ${required}; run tools/fetch_native_deps.py")
  endif()
endforeach()
set(SAMPS_RUNTIME_FILES
  "${SAMPS_ORT_ROOT}/runtimes/win-x64/native/onnxruntime.dll|samps_onnxruntime.dll"
  "${SAMPS_ORT_ROOT}/runtimes/win-x64/native/onnxruntime_providers_shared.dll|onnxruntime_providers_shared.dll"
  "${SAMPS_DML_ROOT}/bin/x64-win/DirectML.dll|DirectML.dll")
# Copies the runtime next to a target so it can be loaded from the module's own directory.
function(samps_stage_runtime target)
  foreach(entry ${SAMPS_RUNTIME_FILES})
    string(REPLACE "|" ";" pair "${entry}")
    list(GET pair 0 source)
    list(GET pair 1 name)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${source}" "$<TARGET_FILE_DIR:${target}>/${name}")
  endforeach()
endfunction()

add_library(samps_core STATIC native/src/engine.cpp native/src/runtime.cpp native/src/resample.cpp
  native/src/yoloe26.cpp native/src/sam3.cpp)
target_include_directories(samps_core PUBLIC native/include "${CMAKE_SOURCE_DIR}/.deps/include")
target_include_directories(samps_core PRIVATE native/src "${SAMPS_ORT_ROOT}/build/native/include" "${SAMPS_DML_ROOT}/include")
if(WIN32)
  target_compile_definitions(samps_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(samps_core PUBLIC bcrypt dxgi)
endif()
if(MSVC)
  target_compile_options(samps_core PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(samps_validation native/tests/validation.cpp)
  target_link_libraries(samps_validation PRIVATE samps_core)
  add_executable(samps_behavior native/tests/behavior.cpp)
  target_link_libraries(samps_behavior PRIVATE samps_core)
  add_executable(samps_prompt native/tests/prompt.cpp)
  target_link_libraries(samps_prompt PRIVATE samps_core)
  add_test(NAME prompt_geometry COMMAND samps_prompt)
  foreach(tool samps_validation samps_behavior samps_prompt)
    target_compile_options(${tool} PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    samps_stage_runtime(${tool})
  endforeach()
  if(WIN32)
    target_link_libraries(samps_behavior PRIVATE psapi)
  endif()
endif()
set(UXP_SDK_ROOT "" CACHE PATH "External Adobe Hybrid SDK root, containing src/api")
option(SAMPS_BUILD_HYBRID "Build the UXP Hybrid plugin (default product)" ON)
if(SAMPS_BUILD_HYBRID)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The UXP Hybrid binary currently targets Windows x64")
  endif()
  if(NOT EXISTS "${UXP_SDK_ROOT}/src/api/UxpAddonShared.h" OR
     NOT EXISTS "${UXP_SDK_ROOT}/src/utilities/UxpAddon.cpp")
    message(FATAL_ERROR "Set UXP_SDK_ROOT to an Adobe UXP Hybrid SDK root containing src/api and src/utilities; use SAMPS_BUILD_HYBRID=OFF for core-only builds")
  endif()
  add_library(samps SHARED native/src/addon.cpp "${UXP_SDK_ROOT}/src/utilities/UxpAddon.cpp")
  target_include_directories(samps PRIVATE "${UXP_SDK_ROOT}/src/api" "${UXP_SDK_ROOT}/src/utilities")
  target_link_libraries(samps PRIVATE samps_core)
  target_compile_options(samps PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  set_target_properties(samps PROPERTIES OUTPUT_NAME "samps-${PROJECT_VERSION}" SUFFIX ".uxpaddon" PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist/win/x64")
  samps_stage_runtime(samps)
endif()
